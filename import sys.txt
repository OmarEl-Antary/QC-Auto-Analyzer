def send_telegram(message):
    try:
        url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"
        response = requests.post(url, json={
            "chat_id": MY_CHAT_ID,
            "text": message
        })
        print(f"Telegram: {response.json().get('ok')}")
    except Exception as e:
        print(f"Telegram Error: {e}")

def process_story(issue):
    story_key = issue.key
    summary = issue.fields.summary

    try:
        print(f"\nProcessing: {story_key} - {summary}")

        chat_completion = client.chat.completions.create(
            messages=[
                {
                    "role": "system",
                    "content": "You are a Senior Software QC Specialist. Write professional Test Cases in Arabic with (Step / Expected Result) format."
                },
                {
                    "role": "user",
                    "content": f"Analyze this Story: {summary}"
                }
            ],
            model="llama-3.1-8b-instant",
        )

        ai_test_cases = chat_completion.choices[0].message.content

        jira.add_comment(story_key, f"AI Analysis (Groq):\n\n{ai_test_cases}")
        print(f"Done: {story_key}")

        send_telegram(f"✅ تم تحليل {story_key}\n📝 {summary}")

        time.sleep(2)

    except Exception as e:
        print(f"FAILED {story_key}: {str(e)}")
        send_telegram(f"❌ فشل تحليل {story_key}: {str(e)}")

def run_jql(jql_query):
    print(f"Running JQL: {jql_query}")

    issues = jira.search_issues(jql_query, maxResults=50)
    total = len(issues)
    print(f"Found {total} Stories")

    if total == 0:
        print("No stories found!")
        send_telegram("⚠️ مفيش Stories اتلاقت بالـ JQL ده!")
        return

    send_telegram(f"🚀 بدأ التحليل!\nعدد الـ Stories: {total}")

    for i, issue in enumerate(issues, 1):
        print(f"\n[{i}/{total}]")
        process_story(issue)

    send_telegram(f"🎉 خلص التحليل!\nتم تحليل {total} Stories بنجاح في جيرا.")
    print("\nALL DONE!")
